cmake_minimum_required(VERSION 3.20)
project(WSE_parallel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(GFSIM_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.gfsim-parallel-src" CACHE PATH "Pinned independent GFSIM checkout")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GFSIM_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE GFSIM_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT GFSIM_REV STREQUAL "cc833bf4003aac17f0177067e70ecaffaf0d54c3")
  message(FATAL_ERROR "GFSIM revision differs from dependency.lock.json: ${GFSIM_REV}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GFSIM_SOURCE_DIR}" diff HEAD --exit-code RESULT_VARIABLE GFSIM_DIRTY OUTPUT_QUIET)
if(NOT GFSIM_DIRTY EQUAL 0)
  message(FATAL_ERROR "GFSIM tracked source must be clean")
endif()
add_subdirectory("${GFSIM_SOURCE_DIR}" gfsim EXCLUDE_FROM_ALL)
add_library(wse_parallel src/model.cpp)
target_include_directories(wse_parallel PUBLIC include)
target_link_libraries(wse_parallel PUBLIC gfsim)
target_compile_options(wse_parallel PRIVATE -Wall -Wextra -Wpedantic)
add_executable(wse_parallel_runner src/main.cpp)
target_link_libraries(wse_parallel_runner PRIVATE wse_parallel)
include(CTest)
find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
add_test(NAME wse_parallel_prototype COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/verify.py" --runner $<TARGET_FILE:wse_parallel_runner> --output "${CMAKE_CURRENT_BINARY_DIR}/prototype-evidence" --stage prototype)
add_test(NAME wse_parallel_40 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/verify.py" --runner $<TARGET_FILE:wse_parallel_runner> --output "${CMAKE_CURRENT_BINARY_DIR}/mesh40-evidence" --stage mesh40)
set_tests_properties(wse_parallel_40 PROPERTIES DEPENDS wse_parallel_prototype)
